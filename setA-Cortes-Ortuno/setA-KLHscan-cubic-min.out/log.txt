//output directory: setA-KLHscan-cubic-min.out/
a := 1e-9
R_out := 40e-9
R_in := 32e-9
Nx := 96
Ny := 96
Nz := 128
SetGridsize(Nx, Ny, Nz)
SetCellsize(a, a, a)
EdgeSmooth = 8
Msat = 0.384e6
Aex = 8.78e-12
Dbulk = 0.0
Kc1 = -6e3
Kc2 = 12e3
tableAdd(Kc1)
tableAdd(Kc2)
N_H := 3
N_L := 8
for iH := 0; iH < N_H; iH++ {
	h_val := 0.0
	if iH == 0 {
		h_val = 0.0
	}
	if iH == 1 {
		h_val = 0.2
	}
	if iH == 2 {
		h_val = 0.4
	}
	B_ext = vector(0, 0, h_val)
	for iL := 0; iL < N_L; iL++ {
		Lz_nm := 16 * (iL + 1)
		Lz := 1e-9 * Lz_nm
		outer := Cylinder(2*R_out, Lz)
		inner := Cylinder(2*R_in, Lz)
		annulus := outer.Sub(inner)
		SetGeom(annulus)
		m = randomMag()
		minimize()
		saveas(m, sprint("m_H", iH, "_Lz", Lz_nm, "nm"))
		tablesave()
		print("Done  iH=", iH, "  Hz=", h_val, " T  Lz=", Lz_nm, " nm")
	}
}
