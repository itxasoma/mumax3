//output directory: comparison.out/
a := 1e-9
R_out := 40e-9
R_in := 32e-9
Nx := 96
Ny := 96
Nz := 128
SetGridsize(Nx, Ny, Nz)
SetCellsize(a, a, a)
EdgeSmooth = 8
Msat = 0.384e6
Aex = 8.78e-12
alpha = 0.5
Dbulk = 0.0
Kc1 = -6e3
Kc2 = 12e3
Lz_nm := 64.0
Lz := Lz_nm * a
h_val := 0.2
B_ext = vector(0, 0, h_val)
outer := Cylinder(2*R_out, Lz)
inner := Cylinder(2*R_in, Lz)
annulus := outer.Sub(inner)
SetGeom(annulus)
tableAdd(E_total)
tableAdd(E_exch)
tableAdd(E_anis)
tableAdd(E_demag)
tableAdd(E_zeeman)
m = randomMag()
saveas(m, "m_init")

//=== CASE: cubic + H comparison ===
print("Lz = ", Lz_nm, " nm")
//Lz =  64  nm
print("mu0 H = ", h_val, " T")
//mu0 H =  0.2  T

//=== RUN 1: relax() ===
relax()
saveas(m, "m_relax")

m.LoadFile("comparison.out/m_init.ovf")

//=== RUN 2: minimize() ===
