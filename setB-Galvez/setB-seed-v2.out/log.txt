//output directory: setB-seed-v2.out/
a := 1e-9
R_out := 40e-9
R_in := 32e-9
Nx := 96
Ny := 96
Nz := 128
SetGridsize(Nx, Ny, Nz)
SetCellsize(a, a, a)
EdgeSmooth = 8
Msat = 0.384e6
Aex = 8.78e-12
alpha = 0.5
Dbulk = 2.0e-3
Ku1 = 110e3
AnisU = vector(0, 0, 1)
B_ext = vector(0, 0, 0)
Lz := 96e-9
SetGeom(Cylinder(2*R_out, Lz).Sub(Cylinder(2*R_in, Lz)))
seed_x := R_out / sqrt(2)
seed_y := R_out / sqrt(2)
box_hx := 10e-9
box_hy := 10e-9
core_hz := 12e-9
cap_hz := 8e-9
cap_z0 := core_hz - cap_hz
print("Seed geometry:")
//Seed geometry:
print("  Caixa total: ", 2*core_hz/1e-9, " nm en z")
//  Caixa total:  24  nm en z
print("  Cap +y: z = [", cap_z0/1e-9, ",", core_hz/1e-9, "] nm")
//  Cap +y: z = [ 3.999999999999999 , 12 ] nm
print("  Nucli -z: z = [", -cap_z0/1e-9, ",", cap_z0/1e-9, "] nm  (", 2*cap_z0/1e-9, " nm)")
//  Nucli -z: z = [ -3.999999999999999 , 3.999999999999999 ] nm  ( 7.999999999999998  nm)
print("  Cap -y: z = [", -core_hz/1e-9, ",", -cap_z0/1e-9, "] nm")
//  Cap -y: z = [ -12 , -3.999999999999999 ] nm
pin_r := 2.5e-9
pin_h := 8e-9
tilt := 15.0 * pi / 180.0
pin_ux := sin(tilt) / sqrt(2)
pin_uy := sin(tilt) / sqrt(2)
pin_uz := cos(tilt)
DefRegion(1, Cylinder(2*pin_r, pin_h).Transl(seed_x, seed_y, 0))
Ku1.SetRegion(1, 110e3)
AnisU.SetRegion(1, vector(pin_ux, pin_uy, pin_uz))
m = uniform(0, 0, 1)
m.SetInShape(Cuboid(2*box_hx, 2*box_hy, cap_hz).Transl(seed_x, seed_y, cap_z0+cap_hz/2), Uniform(0, 1, 0))
m.SetInShape(Cuboid(2*box_hx, 2*box_hy, cap_hz).Transl(seed_x, seed_y, -(cap_z0+cap_hz/2)), Uniform(0, -1, 0))
m.SetInShape(Cuboid(2*box_hx, 2*box_hy, 2*cap_z0).Transl(seed_x, seed_y, 0), Uniform(0, 0, -1))
saveas(m.Comp(0), "seed_v2_mx")
saveas(m.Comp(1), "seed_v2_my")
saveas(m.Comp(2), "seed_v2_mz")
print("Fet. Visualitza seed_v2_mz: has de veure blau (−z) al centre, blanc (±y) als extrems.")
